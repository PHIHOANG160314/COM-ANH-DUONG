## 🔴 Task 2: Page Transitions with Framer Motion

### Step 2.1: Update `src/shared/ui/page-transition.tsx`

Replace content with proper framer-motion implementation:

```tsx
import { motion } from "framer-motion";
import { ReactNode } from "react";

interface PageTransitionProps {
  children: ReactNode;
}

export const PageTransition = ({ children }: PageTransitionProps) => (
  <motion.div
    initial={{ opacity: 0, y: 10 }}
    animate={{ opacity: 1, y: 0 }}
    exit={{ opacity: 0, y: -10 }}
    transition={{ duration: 0.25, ease: "easeOut" }}
    style={{ width: "100%" }}
  >
    {children}
  </motion.div>
);
```

### Step 2.2: Update `src/App.tsx`

Wrap the Routes with AnimatePresence:

1. Import `AnimatePresence` from framer-motion
2. Import `useLocation` from react-router-dom
3. Wrap `<Routes>` with `<AnimatePresence mode="wait">`
4. Add `location` and `key` props to Routes

Example pattern:

```tsx
import { AnimatePresence } from "framer-motion";
import { useLocation } from "react-router-dom";

// Inside component:
const location = useLocation();

// Wrap routes:
<AnimatePresence mode="wait">
  <Routes location={location} key={location.pathname}>
    {/* existing routes */}
  </Routes>
</AnimatePresence>;
```
